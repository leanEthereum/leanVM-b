\section{The instruction tables}\label{sec:tables}

Each instruction has its own table.

\subsection{\texttt{XOR}}\label{sec:tab-xor}

Asserts $\loc{o_C}=\loc{o_A}+\loc{o_B}$, the $192$-bit field sum (in $\E$).
\begin{itemize}
\item \textbf{Columns:} $\pc,\fp$; operands $o_A,o_B,o_C$; the two read values $v_A,v_B$ (each three $\K$-limbs); memory counts $r_A,r_B,r_C$; bytecode count $r_{\mathrm{bc}}$.
\item \textbf{Constraints:} none.
\item \textbf{Flushes:}
  \begin{itemize}
  \item \textbf{State:} \pull\ $\tup{\dsep{ST},\pc,\fp}$, \push\ $\tup{\dsep{ST},\gen\cdot\pc,\fp}$.
  \item \textbf{Bytecode:} \pull\ $\tup{\dsep{BC},\pc,r_{\mathrm{bc}},\opc{XOR},o_A,o_B,o_C,0,0}$, \push\ the same at $\gen\cdot r_{\mathrm{bc}}$.
  \item \textbf{Memory} ($X\in\{A,B\}$): \pull\ $\tup{\dsep{MEM},\fp\cdot o_X,r_X,v_X}$, \push\ $\tup{\dsep{MEM},\fp\cdot o_X,\gen\cdot r_X,v_X}$; and for the destination, the same pair at $\fp\cdot o_C$ with the three value coordinates $v_{A,i}+v_{B,i}$, of degree one.
  \end{itemize}
\end{itemize}

\subsection{\texttt{MUL\_NATIVE}}\label{sec:tab-mul}

Asserts $\loc{o_C}=\loc{o_A} \cdot \loc{o_B}$, the $192$-bit field multiplication (in $\E$).
\begin{itemize}
\item \textbf{Columns:} $\pc,\fp$; operands $o_A,o_B,o_C$; the two read values $v_A,v_B$ (each three $\K$-limbs); memory counts $r_A,r_B,r_C$; bytecode count $r_{\mathrm{bc}}$.
\item \textbf{Constraints:} none.
\item \textbf{Flushes:}
  \begin{itemize}
  \item \textbf{State:} \pull\ $\tup{\dsep{ST},\pc,\fp}$, \push\ $\tup{\dsep{ST},\gen\cdot\pc,\fp}$.
  \item \textbf{Bytecode:} \pull\ $\tup{\dsep{BC},\pc,r_{\mathrm{bc}},\opc{MUL},o_A,o_B,o_C,0,0}$, \push\ the same at $\gen\cdot r_{\mathrm{bc}}$.
  \item \textbf{Memory} ($X\in\{A,B\}$): \pull\ $\tup{\dsep{MEM},\fp\cdot o_X,r_X,v_X}$, \push\ $\tup{\dsep{MEM},\fp\cdot o_X,\gen\cdot r_X,v_X}$; and for the destination, the same pair at $\fp\cdot o_C$ with the three value coordinates carrying the product $v_Av_B$ in $\E$, of degree two. With $y^3=y+1$, put
    \[
    p_0=v_{A,0}v_{B,0},\quad
    p_1=v_{A,0}v_{B,1}+v_{A,1}v_{B,0},\quad
    p_2=v_{A,0}v_{B,2}+v_{A,1}v_{B,1}+v_{A,2}v_{B,0},
    \]
    \[
    p_3=v_{A,1}v_{B,2}+v_{A,2}v_{B,1},\qquad
    p_4=v_{A,2}v_{B,2};
    \]
    the coordinates are then $p_0+p_3$, $p_1+p_3+p_4$ and $p_2+p_4$, twelve products over the nine pairs $v_{A,i}v_{B,j}$.
  \end{itemize}
\end{itemize}

\subsection{\texttt{SET\_CONSTANT}}\label{sec:tab-set}

Asserts $\loc{o}=k$ for some constant $k \in \E$ (which is how a constant enters the read-only memory).
\begin{itemize}
\item \textbf{Columns:} $\pc,\fp$; operand $o$; immediate $k$ ($192$-bit, three $\K$-limbs $k_0,k_1,k_2$); memory count $r$; bytecode count $r_{\mathrm{bc}}$.
\item \textbf{Constraints:} none.
\item \textbf{Flushes:}
  \begin{itemize}
  \item \textbf{State:} \pull\ $\tup{\dsep{ST},\pc,\fp}$, \push\ $\tup{\dsep{ST},\gen\cdot\pc,\fp}$.
  \item \textbf{Bytecode:} \pull\ $\tup{\dsep{BC},\pc,r_{\mathrm{bc}},\opc{SET},o,k_0,k_1,k_2,0}$, \push\ the same at $\gen\cdot r_{\mathrm{bc}}$ 
  \item \textbf{Memory:} \pull\ $\tup{\dsep{MEM},\fp\cdot o,r,k}$, \push\ $\tup{\dsep{MEM},\fp\cdot o,\gen\cdot r,k}$ 
  \end{itemize}
\end{itemize}

\subsection{\texttt{DEREF}}\label{sec:tab-deref}

\texttt{DEREF} follows a pointer and asserts the cell it points to. The row reads three cells:
\begin{itemize}
\item the \emph{pointer} $p$, at $\fp\cdot\alpha$;
\item the \emph{target} $v_2$, at the dereferenced address $p\cdot\beta$;
\item the \emph{local} value $v_3$, at $\fp\cdot\gamma$.
\end{itemize}
The store asserts that $v_2$ equals the source chosen by the mode $s$ (\S\ref{sec:vm}): the local cell $v_3$, the return address $\gen^{2}\cdot\pc$, or the frame pointer $\fp$. The mode is two boolean flags $(f_{\pc},f_{\fp})$: $(0,0)$ for $\texttt{deref\_cell}[\gamma]$, $(1,0)$ for $\texttt{deref\_pc}$, $(0,1)$ for $\texttt{deref\_fp}$.
\begin{itemize}
\item \textbf{Columns:} $\pc,\fp$; operands $\alpha,\beta,\gamma$; flags $f_{\pc},f_{\fp}$; pointer $p$ (a single $\K$-limb); the local value $v_3$ (three $\K$-limbs); memory counts $r_1,r_2,r_3$; bytecode count $r_{\mathrm{bc}}$.
\item \textbf{Constraints:} none.
\item \textbf{Flushes:}
  \begin{itemize}
  \item \textbf{State:} \pull\ $\tup{\dsep{ST},\pc,\fp}$, \push\ $\tup{\dsep{ST},\gen\cdot\pc,\fp}$.
  \item \textbf{Bytecode:} \pull\ $\tup{\dsep{BC},\pc,r_{\mathrm{bc}},\opc{DRF},\alpha,\beta,\gamma,f_{\pc},f_{\fp}}$, \push\ the same at $\gen\cdot r_{\mathrm{bc}}$.
  \item \textbf{Memory:} three reads, each \pull\ at count $r$ and \push\ at $\gen\cdot r$: pointer $\tup{\dsep{MEM},\fp\cdot\alpha,r_1,p,0,0}$, local cell $\tup{\dsep{MEM},\fp\cdot\gamma,r_3,v_3}$, and the target $\tup{\dsep{MEM},p\cdot\beta,r_2,v_2}$ with
    \[
      v_2 \;=\; (1+f_{\pc}+f_{\fp})\,v_3 \;+\; f_{\pc}\,(\gen^{2}\cdot\pc) \;+\; f_{\fp}\,\fp
    \]
    as its three value coordinates, which give $v_3$, $\gen^{2}\cdot\pc$, or $\fp$ at the three flag settings (in characteristic $2$ the $v_3$ term zeroes for the last two). Only the low limb takes the two $\K$-valued sources; the upper two are the gated limbs of $v_3$ alone.
  \end{itemize}
\end{itemize}

\subsection{\texttt{JUMP}}\label{sec:tab-jump}

A conditional jump on whether the condition $c=\loc{o_c}$ is nonzero. The condition must be $\K$-valued. It transfers to $\pc\gets d$, $\fp\gets f$ when $c\neq0$ (the destination $d=\loc{o_d}$, the updated program counter, and the updated frame pointer $f=\loc{o_f}$), and falls through, $\pc\gets\gen\cdot\pc$, $\fp\gets\fp$, when $c=0$. The branch is taken according to a boolean indicator column $b=[\,c\neq0\,]$ (Definition~\ref{def:iverson}), which a prover-supplied inverse $w$ certifies through the two degree-$2$ constraints
\[
  b=c\cdot w,\qquad c\,(b+1)=0 .
\]
When $c\neq0$ the second forces $b=1$ and the first then $w=c^{-1}$; when $c=0$ the first forces $b=0$. Writing $s=\gen\cdot\pc$ for the fall-through, the next state is the selection
\[
  \nxt{\pc}=b\cdot d+(b+1)\,s,\qquad \nxt{\fp}=b\cdot f+(b+1)\,\fp,
\]
\begin{itemize}
\item \textbf{Columns:} $\pc,\fp$; operands $o_c,o_d,o_f$; the condition $c$, the destination $d$ and the frame $f$ (one $\K$-limb each); inverse $w$ (one $\K$-limb, $c^{-1}$); indicator $b$ (one $\K$-limb, $0/1$); memory counts $r_c,r_d,r_f$; bytecode count $r_{\mathrm{bc}}$.
\item \textbf{Constraints:}  $b=c\cdot w$ and $c\,(b+1)=0$
\item \textbf{Flushes:}
  \begin{itemize}
  \item \textbf{State:} \pull\ $\tup{\dsep{ST},\pc,\fp}$, \push\ $\tup{\dsep{ST},b\,d+b\,(\gen\cdot\pc)+\gen\cdot\pc,\;b\,f+b\,\fp+\fp}$.
  \item \textbf{Bytecode:} \pull\ $\tup{\dsep{BC},\pc,r_{\mathrm{bc}},\opc{JMP},o_c,o_d,o_f,0,0}$, \push\ the same at $\gen\cdot r_{\mathrm{bc}}$.
  \item \textbf{Memory:} three reads, each \pull\ at count $r$ and \push\ at $\gen\cdot r$: $\tup{\dsep{MEM},\fp\cdot o_c,r_c,c,0,0}$, $\tup{\dsep{MEM},\fp\cdot o_d,r_d,d,0,0}$ and $\tup{\dsep{MEM},\fp\cdot o_f,r_f,f,0,0}$.
  \end{itemize}
\end{itemize}

\subsection{\texttt{PACK64X2}}\label{sec:tab-pack64x2}

Asserts that two source cells are $\K$-valued and packs them into one canonical $128$-bit word.
\begin{itemize}
\item \textbf{Columns:} $\pc,\fp$; operands $o_A,o_B,o_C$; source limbs $v_A,v_B\in\K$; memory counts $r_A,r_B,r_C$; bytecode count $r_{\mathrm{bc}}$.
\item \textbf{Constraints:} none. 
\item \textbf{Flushes:}
  \begin{itemize}
  \item \textbf{State:} \pull\ $\tup{\dsep{ST},\pc,\fp}$, \push\ $\tup{\dsep{ST},\gen\cdot\pc,\fp}$.
  \item \textbf{Bytecode:} \pull\ $\tup{\dsep{BC},\pc,r_{\mathrm{bc}},\opc{PACK},o_A,o_B,o_C,0,0}$, \push\ the same at $\gen\cdot r_{\mathrm{bc}}$.
  \item \textbf{Memory} ($X\in\{A,B\}$): \pull\ $\tup{\dsep{MEM},\fp\cdot o_X,r_X,v_X,0,0}$, \push\ $\tup{\dsep{MEM},\fp\cdot o_X,\gen\cdot r_X,v_X,0,0}$; and for the destination, \pull\ $\tup{\dsep{MEM},\fp\cdot o_C,r_C,v_A,v_B,0}$, \push\ $\tup{\dsep{MEM},\fp\cdot o_C,\gen\cdot r_C,v_A,v_B,0}$. The literal zeros are what make the two source range assertions and the destination packing exact.
  \end{itemize}
\end{itemize}

\subsection{\texttt{BLAKE2S}}\label{sec:tab-blake2s}

Compresses a $64$-byte message block (four $128$-bit chunks) under a $32$-byte chaining value to the $32$-byte result at $c$. Each chunk occupies one $192$-bit cell through the canonical embedding with top limb zero. The message chunks are addressed independently at $\fp\cdot o_i$, while $\fp\cdot o_V$ is the base of the two consecutive chaining-value cells $\fp\cdot o_V,\fp\cdot o_V \cdot \gen$. The output similarly occupies $\fp\cdot o_C$ and $\fp\cdot o_C \cdot \gen$, so each row accesses eight cells. The $128$-bit immediate $\md$ packs the $64$-bit counter, $32$-bit block length, and $32$-bit flags required by the BLAKE2s specification; its two $\K$-lanes $\md_0,\md_1$ ride the bytecode tuple.

Flock~\cite{flock-impl} proves the compression relation with a batched R1CS over a Boolean witness (Annex~\ref{annex:flock}). The witness is packed into $\qflock$, with $64$ bits per $\K$-element.

The message, result, chaining value, and metadata are all contained in $\qflock$. Each $64$-bit Flock word occupies one packed slot, and each BLAKE2s cell contributes its two low limbs; the top limb is canonically zero. The eighteen limbs $v_{A0},\dots,v_{C3},v_{V0},\dots,v_{V3},\md_0,\md_1$ therefore need no separate commitment: each is a strided point evaluation of $\qflock$. Linking these embedded values to the full memory interaction implicitly forces the top limb of every BLAKE2s-accessed cell to zero, and the zero limbs are never committed. The generalized R1CS matrices take the chaining value, counter, block length, and flags as free input rows; the bus and bytecode interactions bind them.
\begin{itemize}
\item \textbf{Columns:} $\pc,\fp$; operands $o_0,o_1,o_2,o_3,o_V,o_C$; per-cell memory counts $r_0,\dots,r_3$ (the four message cells), $r_{V0},r_{V1}$ (the two chaining-value cells), and $r_{C0},r_{C1}$ (the two output cells); bytecode count $r_{\mathrm{bc}}$. The eighteen value limbs above are committed "inside" $\qflock$, see above; each cell's third limb is zero. The eight addresses are the products $\gen^{k}\fp\cdot o$, $k\in\{0,1\}$.
\item \textbf{Constraints:} none. The compression relation is handled by flock.
\item \textbf{Flushes:}
  \begin{itemize}
  \item \textbf{State:} \pull\ $\tup{\dsep{ST},\pc,\fp}$, \push\ $\tup{\dsep{ST},\gen\cdot\pc,\fp}$.
  \item \textbf{Bytecode:} \pull\ $\tup{\dsep{BC},\pc,r_{\mathrm{bc}},\opc{B3},o_0,o_1,o_2,o_3,o_V,o_C,\md_0,\md_1}$, \push\ the same at $\gen\cdot r_{\mathrm{bc}}$.
  \item \textbf{Memory:} eight reads, each \pull\ at count $r$ and \push\ at $\gen\cdot r$, its embedded value carrying two limbs from $\qflock$ and a zero top limb (\S\ref{sec:memchan}): four independently addressed message cells, the chaining value at $\fp\cdot o_V,\fp\cdot o_V \cdot \gen$, and the result at $\fp\cdot o_C,\fp\cdot o_C \cdot \gen$.
  \end{itemize}
\end{itemize}
