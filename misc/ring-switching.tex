\begin{center}
How to transform a PCS for $\F_{2^f}$ into a PCS for $\F_{2^e}$, for arbitrary $e$ and $f$.
\end{center}

\paragraph{Goal.} We want $f$ \emph{bit}-valued multilinears $P_0,\dots,P_{f-1}:\{0,1\}^m\to\{0,1\}$ over the same $m$ variables, to be committed through a dense PCS over the \emph{packing} field $\F_{2^f}$ (if we can commit to bits, we can trivially commit in any extension). The evaluation claims, however, arrive over a \emph{different} field: at an opening point $r\in\F_{2^e}^m$ we are given $P_i(r)=\alpha_i\in\F_{2^e}$. Ring switching~\cite{DP26} packs the $f$ polynomials into one $\F_{2^f}$-dense polynomial and reduces the $f$ claims to a single claim on it. The two extension degrees, $f$ (packing) and $e$ (opening), are unrelated (not necessarily a sub-field / extension-field).

\paragraph{Packing.} Fix an $\F_2$-basis $\{b^{F}_i\}_{i=0}^{f-1}$ of $\F_{2^f}$. Consider the multilinear polynomial:
\[
P_\pkd(y)=\sum_{i=0}^{f-1} P_i(y)\,b^{F}_i,\qquad y\in\{0,1\}^m,
\]
dense over $\F_{2^f}$, which we commit to.

\paragraph{Reduction.} The claims share a \emph{common} point $r\in\F_{2^e}^m$; each is
\[
\alpha_i=P_i(r)=\sum_{y\in\{0,1\}^m}\eq(r,y)\,P_i(y),
\]
with bits $P_i(y)$ and weights $\eq(r,y)\in\F_{2^e}$. Fix an $\F_2$-basis $\{b^{E}_w\}_{w=0}^{e-1}$ of the \emph{opening} field $\F_{2^e}$ and expand the weights, $\eq(r,y)=\sum_{w=0}^{e-1} A(y,w)\,b^{E}_w$, with $A:\{0,1\}^m\times[e]\to\F_2$ public, multilinear in $y$ for each fixed $w$, with efficiently computable MLE (see \nameref{section:eval_B} below). Substituting and pulling out the basis,
\[
\alpha_i=\sum_{w=0}^{e-1}b^{E}_w\underbrace{\sum_{y\in\{0,1\}^m}A(y,w)\,P_i(y)}_{\in\,\F_2},
\]
so the $w$-th $\F_2$-coordinate of $\alpha_i$ is $\alpha_{w,i}=\sum_{y\in\{0,1\}^m}A(y,w)\,P_i(y)$. Now recombine \emph{across $i$} with the \emph{packing} basis $\{b^{F}_i\}$: define, for each $w\in\{0,\dots,e-1\}$,
\[
t_w:=\sum_{i=0}^{f-1}\alpha_{w,i}\,b^{F}_i=\sum_{y\in\{0,1\}^m}A(y,w)\underbrace{\sum_{i=0}^{f-1}P_i(y)\,b^{F}_i}_{=\,P_\pkd(y)}=\sum_{y\in\{0,1\}^m}A(y,w)\,P_\pkd(y)\ \in\ \F_{2^f}.
\]
These are $e$ claims about the packed polynomial $P_\pkd$. Let $\ell=\lceil\log_2 e\rceil$, identify each $w<e$ with its $\ell$-bit encoding $\bar w\in\{0,1\}^{\ell}$, and pad $t_w=A(y,w)=0$ for $e\le w<2^\ell$. The verifier samples a multilinear batching point $s\leftarrow\F_{2^f}^{\ell}$ and uses weights $\lambda_w=\eq(s,\bar w)$. One sumcheck proves
\[
\sum_{y\in\{0,1\}^m}B_s(y)\,P_\pkd(y)=T(s),\qquad B_s(y):=\sum_{w<2^\ell}\eq(s,\bar w)A(y,w),\qquad T(s):=\sum_{w<2^\ell}\eq(s,\bar w)t_w.
\]
Its output is the packed claim $P_\pkd(r')=\alpha'$ at a point $r'\in\F_{2^f}^m$; the verifier computes $B_s(r')$ itself. This is the complete reduction.

\paragraph{Soundness.} Suppose the input is wrong: $\alpha_i\neq P_i(r)$ for some $i$, against the committed $P_\pkd$. Then the reduction outputs a correct packed claim $P_\pkd(r')=\alpha'$ with probability at most
\[
\underbrace{\frac{\ell}{2^f}}_{\text{batching}}\;+\;\underbrace{\frac{2m}{2^f}}_{\text{sumcheck}}\;=\;\frac{2m+\ell}{2^f}.
\]
\begin{enumerate}
\item The map $(\alpha_i)_{i=0}^{f-1}\mapsto(t_w)_{w=0}^{e-1}$ is a \emph{bijection} $\F_{2^e}^f\to\F_{2^f}^e$. Therefore a wrong input forces $t_w\neq\sum_yA(y,w)P_\pkd(y)$ for at least one $w$.
\item The multilinear extension of these coordinate errors is nonzero because it is nonzero at that Boolean point $\bar w$. By Schwartz--Zippel, its evaluation at $s\leftarrow\F_{2^f}^{\ell}$ vanishes with probability at most $\ell/2^f$.
\item Sumcheck started from this false value (summand of degree $2$, over $m$ variables) ends at a correct claim $P_\pkd(r')=\alpha'$ with probability $\le 2m/2^f$.
\end{enumerate}

\section*{Evaluating $B$}
\label{section:eval_B}

It remains to compute $B_s(r')$. The value $A(y,w)\in\F_2$ is the $w$-th coordinate of $\eq(r,y)$, i.e.\ $\eq(r,y)=\sum_{w=0}^{e-1}A(y,w)b^E_w$. Following~\cite{Flock}, define the $\F_2$-linear map $\Phi_s:\F_{2^e}\to\F_{2^f}$ by $\Phi_s(b^E_w)=\eq(s,\bar w)$. For every $y\in\{0,1\}^m$:
\[
\Phi_s\big(\eq(r,y)\big)=\Phi_s\Big(\sum_{w=0}^{e-1}A(y,w)b^E_w\Big)=\sum_{w=0}^{e-1}A(y,w)\eq(s,\bar w)=B_s(y).
\]
Thus, on the hypercube, $B_s$ is ``compute $\eq(r,y)\in\F_{2^e}$, then apply $\Phi_s$.''

\paragraph{Closed form.} Two elementary facts do all the work.

\emph{Fact 1 ($\eq$ in characteristic $2$).} In each factor of $\eq$ the two $x_jy_j$ terms cancel:
\[
\eq(x,y)=\prod_{j=1}^m\bigl(x_jy_j+(1+x_j)(1+y_j)\bigr)=\prod_{j=1}^m(1+x_j+y_j).
\]

\emph{Fact 2 (linear maps are sums of squarings).} View $\Phi_s$ as mapping into $L=\F_{2^{\mathrm{lcm}(e,f)}}$, the smallest field containing both $\F_{2^e}$ and $\F_{2^f}$ (for $e=f$: $L=\F_{2^e}$, nothing to embed). Squaring is additive in characteristic $2$, so any expression
\[
a\;\mapsto\;c_0\,a+c_1\,a^2+c_2\,a^4+\dots+c_{e-1}\,a^{2^{e-1}},\qquad c_k\in L,
\]
is an $\F_2$-linear map $\F_{2^e}\to L$, and every such map can be written this way. We find the $c_k$ by writing out what $\Phi_s$ must do on the basis:
\[
\sum_{k=0}^{e-1}\bigl(b^E_w\bigr)^{2^k}c_k=\eq(s,\bar w),\qquad w=0,\dots,e-1.
\]
This square system is invertible: a combination that vanishes on the basis vanishes on all $2^e$ field elements, whereas its degree is at most $2^{e-1}$, so all its coefficients must be zero. Its matrix depends only on the basis and can therefore be inverted once; each opening then computes $(c_k)_{k<e}$ by an $O(e^2)$ matrix-vector product.

Now compute. Squaring respects sums and products and fixes bits ($y_j^2=y_j$), so on the hypercube, by Fact~1,
\[
\eq(r,y)^{2^k}=\prod_{j=1}^m(1+r_j+y_j)^{2^k}=\prod_{j=1}^m\bigl(1+r_j^{2^k}+y_j\bigr)=\eq\bigl(r^{2^k},y\bigr),
\qquad r^{2^k}:=\bigl(r_1^{2^k},\dots,r_m^{2^k}\bigr).
\]
Since $B_s$ is multilinear, evaluating it at $r'$ and expanding with Fact~2:
\[
B_s(r')=\sum_{y\in\{0,1\}^m}B_s(y)\,\eq(r',y)
=\sum_{y\in\{0,1\}^m}\Phi_s\bigl(\eq(r,y)\bigr)\,\eq(r',y)
=\sum_{k<e}c_k\sum_{y\in\{0,1\}^m}\eq\bigl(r^{2^k},y\bigr)\,\eq(r',y).
\]
The inner sum evaluates, at $r'$, the multilinear extension of $y\mapsto\eq(r^{2^k},y)$, which is $\eq(r^{2^k},\cdot\,)$ itself. So it collapses to $\eq(r^{2^k},r')$, and by Fact~1 again:
\[
\boxed{\ \
B_s(r') \;=\; \sum_{k<e}\, c_k \prod_{j=1}^m
\bigl( 1 + r_j^{2^k} + r'_j \bigr).
\ \ }
\]

\paragraph{Cost.} Level $k$ needs one squaring per coordinate (to get $r_j^{2^k}$ from $r_j^{2^{k-1}}$) and one $m$-factor product: $O(em)$ multiplications in $L$, on top of the generic $O(e^2)$ construction of the $c_k$.

\section*{Instantiation: $64$-bit packing, $192$-bit openings}

The recursion verifier of leanVM-b runs the reduction with $f=64$ and $e=192$:
\[
K=\F_{2^{64}}=\F_2[X]/(q),\quad q=X^{64}+X^4+X^3+X+1,
\qquad
\E=\F_{2^{192}}=K[\Theta]/(\Theta^3+\Theta+1).
\]
Write $x$ and $\vartheta$ for the residues of $X$ and $\Theta$. The packing basis is $b^F_i=x^i$, and the opening basis is the product basis
\[
b^E_{64\nu+i}=x^i\vartheta^{\nu},\qquad i<64,\quad \nu<3,
\]
so a committed word holds the $64$ bits $P_0(y),\dots,P_{63}(y)$. Since $K\subset\E$, the field $L$ of Fact~2 is $\E$ itself and the coefficients $c_k$ lie in $\E$.

One departure from the generic setup: the $\alpha_i$ are not handed to the verifier. The incoming claim is a fixed $\E$-linear combination of them with public weights, so the prover transmits the $64$ values $\alpha_i$ and the verifier recomputes that combination from them.

This verifier runs inside a VM where every field multiplication is an instruction, so we specialize two parts of the reduction. Throughout, $\Phi$, $B$ and $T$ carry the dependence on $\rho$ that $\Phi_s$, $B_s$ and $T(s)$ carried on $s$.

\paragraph{One challenge, no sumcheck of its own.} The generic reduction batches with an $\eq$ tensor over $\ell=\lceil\log_2e\rceil$ variables and proves the batched claim by its own sumcheck. We instead draw a single $\rho\leftarrow\E$ once the verifier holds the $\alpha_i$, and replace the weights $\eq(s,\bar w)$ by weights polynomial in $\rho$, fixed in \eqref{eq:phi-choice}. The batched claim $\sum_yB(y)P_\pkd(y)=T$ then has exactly the shape the PCS opening already proves: a transparent $\E$-valued weight against the committed $K$-valued $P_\pkd$. We fold it into that opening, so the reduction adds no sumcheck and no second opening.

\paragraph{Fix the map, not the weights.} The weights and the map determine each other, $\lambda_w=\Phi(b^E_w)$, so one may fix either. Fix the map. The in-VM verifier never evaluates a single $\lambda_w$; it applies $\Phi$, at the two places below, and each application costs one squaring and one multiplication per nonzero coefficient. The $\lambda_w$ are needed only outside the VM, where they are built once, off the hot path. The cost of the reduction is therefore the size of the support of $\Phi$, and we take that support to be $64$:
\begin{equation}
\Phi(a)=\sum_{k<64}\rho^{\,k}a^{2^k},
\qquad
\lambda_w=\Phi\bigl(b^E_w\bigr).
\label{eq:phi-choice}
\end{equation}
Its coefficients are the successive powers of $\rho$: $63$ multiplications and no table. The obvious choice $\lambda_w=\rho^w$ makes $\Phi$ dense instead, with all $192$ coefficients nonzero, and the verifier must then recover them by the $O(e^2)$ solve of the previous section. The support cannot be made smaller: $64$ terms is the minimum for any sound $\Phi$, as shown below.

\paragraph{Where $\Phi$ is applied.} First, the target $T$. The verifier has the $\alpha_i$ and never needs to form the $t_w$:
\[
T=\sum_{w<192}\lambda_w t_w
=\sum_{w<192}\Phi\bigl(b^E_w\bigr)\sum_{i<64}\alpha_{w,i}\,x^i
=\sum_{i<64}x^i\,\Phi\Bigl(\sum_{w<192}\alpha_{w,i}b^E_w\Bigr)
=\sum_{i<64}x^i\,\Phi(\alpha_i),
\]
using $\alpha_{w,i}\in\F_2$ and the $\F_2$-linearity of $\Phi$. The VM computes the right-hand side: $64$ applications of $\Phi$, and no transpose of the bit matrix $(\alpha_{w,i})$.

Second, the weight at the PCS query point $r'$, which is the boxed formula of the previous section with the coefficients of \eqref{eq:phi-choice}:
\[
B(r')=\sum_{k<64}\rho^{\,k}\prod_{j=1}^{m}\bigl(1+r_j^{2^k}+r'_j\bigr).
\]

\paragraph{Soundness of the batching.} Suppose the transmitted $\alpha_i$ are wrong. Under the bijection $\F_{2^e}^f\to\F_{2^f}^e$ of Soundness item~1, their error maps to a nonzero $\Delta\in K^{192}$, fixed before $\rho$ is drawn. The check passes only if
\[
\sum_{w<192}\lambda_w\Delta_w=\sum_{k<64}\rho^{\,k}W_k=0,
\qquad
W_k:=\sum_{w<192}\bigl(b^E_w\bigr)^{2^k}\Delta_w.
\]
Write $w=64\nu+i$ and $u_k:=\vartheta^{2^k}$, so that $\bigl(b^E_w\bigr)^{2^k}=x^{i2^k}u_k^{\nu}$. Collecting the factors that lie in $K$,
\[
W_k=\sum_{\nu<3}u_k^{\nu}R_{\nu,k},
\qquad
R_{\nu,k}:=\sum_{i<64}x^{i2^k}\Delta_{64\nu+i}\ \in K.
\]
Two observations finish it.
\begin{enumerate}
\item $u_k\notin K$. The elements of $K$ are the fixed points of $a\mapsto a^{2^{64}}$, so $u_k\in K$ would give $\vartheta^{2^{k+64}}=\vartheta^{2^k}$, and applying the inverse Frobenius $k$ times gives $\vartheta^{2^{64}}=\vartheta$, that is $\vartheta\in K$, contradicting $[\E:K]=3$. As that degree is prime, any element outside $K$ generates $\E$ over $K$, so $\{1,u_k,u_k^2\}$ is a $K$-basis of $\E$ and $W_k=0$ forces $R_{0,k}=R_{1,k}=R_{2,k}=0$.
\item For each $\nu$, the coefficient matrix of $\{R_{\nu,k}=0:k<64\}$ is $\bigl(x^{i2^k}\bigr)_{k,i<64}$, the Moore matrix of $x^0,\dots,x^{63}$. A Moore matrix is invertible exactly when the elements it is built from are $\F_2$-independent, and $x^0,\dots,x^{63}$ are an $\F_2$-basis of $K$, so the only solution is $\Delta_{64\nu+i}=0$ for all $i$.
\end{enumerate}
So $\Delta\neq0$ forces $W_k\neq0$ for some $k$, making $\sum_k\rho^kW_k$ a nonzero polynomial in $\rho$ of degree below $64$. By Schwartz--Zippel the error survives with probability at most $63/2^{192}$. The dense weights $\rho^w$ give $191/2^{192}$, since there the quantity checked is $\sum_w\rho^w\Delta_w$ directly, so the sparse map also has the smaller error. Both bounds explain why $\rho$ is drawn from $\E$: over $K$ the first would read $63/2^{64}$, about $58$ bits.

\paragraph{$64$ terms are necessary.} Let $S\subseteq\{0,\dots,191\}$ be any support, so that $\Phi(a)=\sum_{k\in S}c_ka^{2^k}$. The quantity checked above is $\sum_{k\in S}c_kW_k$, so a $\Delta\neq0$ with $W_k=0$ for all $k\in S$ passes for \emph{every} choice of coefficients, with probability $1$. The map $\Delta\mapsto(W_k)_{k\in S}$ is $K$-linear and sends $K^{192}$ into $\E^{|S|}=K^{3|S|}$, so if $|S|<64$ its kernel is nonzero and such a $\Delta$ exists. Hence $|S|\ge64$, and \eqref{eq:phi-choice} attains it.

\paragraph{Cost.} Applying $\Phi$ once costs $63$ squarings and $64$ multiplications, a third of the cost of the dense map, and $B(r')$ costs a further $64$ products of $m$ factors. Replacing the dense map by \eqref{eq:phi-choice} removed $11.9\%$ of the recursion verifier's VM cycles, and deleted the coefficient construction and its baked constants.
