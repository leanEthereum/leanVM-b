\begin{center}
How to transform a PCS for $\F_{2^f}$ into a PCS for $\F_{2^e}$, for arbitrary $e$ and $f$.
\end{center}

\paragraph{Goal.} We want $f$ \emph{bit}-valued multilinears $P_0,\dots,P_{f-1}:\{0,1\}^m\to\{0,1\}$ over the same $m$ variables, to be committed through a dense PCS over the \emph{packing} field $\F_{2^f}$ (if we can commit to bits, we can trivially commit in any extension). The evaluation claims, however, arrive over a \emph{different} field: at an opening point $r\in\F_{2^e}^m$ we are given $P_i(r)=\alpha_i\in\F_{2^e}$. Ring switching~\cite{DP26} packs the $f$ polynomials into one $\F_{2^f}$-dense polynomial and reduces the $f$ claims to a single claim on it. The two extension degrees, $f$ (packing) and $e$ (opening), are unrelated (not necessarily a sub-field / extension-field).

\paragraph{Packing.} Fix an $\F_2$-basis $\{b^{F}_i\}_{i=0}^{f-1}$ of $\F_{2^f}$. Consider the multilinear polynomial:
\[
P_\pkd(y)=\sum_{i=0}^{f-1} P_i(y)\,b^{F}_i,\qquad y\in\{0,1\}^m,
\]
dense over $\F_{2^f}$, which we commit to.

\paragraph{Reduction.} The claims share a \emph{common} point $r\in\F_{2^e}^m$; each is
\[
\alpha_i=P_i(r)=\sum_{y\in\{0,1\}^m}\eq(r,y)\,P_i(y),
\]
with bits $P_i(y)$ and weights $\eq(r,y)\in\F_{2^e}$. Fix an $\F_2$-basis $\{b^{E}_w\}_{w=0}^{e-1}$ of the \emph{opening} field $\F_{2^e}$ and expand the weights, $\eq(r,y)=\sum_{w=0}^{e-1} A(y,w)\,b^{E}_w$, with $A:\{0,1\}^m\times[e]\to\F_2$ public, multilinear in $y$ for each fixed $w$, with efficiently computable MLE (see \nameref{section:eval_B} below). Substituting and pulling out the basis,
\[
\alpha_i=\sum_{w=0}^{e-1}b^{E}_w\underbrace{\sum_{y\in\{0,1\}^m}A(y,w)\,P_i(y)}_{\in\,\F_2},
\]
so the $w$-th $\F_2$-coordinate of $\alpha_i$ is $\alpha_{w,i}=\sum_{y\in\{0,1\}^m}A(y,w)\,P_i(y)$. Now recombine \emph{across $i$} with the \emph{packing} basis $\{b^{F}_i\}$: define, for each $w\in\{0,\dots,e-1\}$,
\[
t_w:=\sum_{i=0}^{f-1}\alpha_{w,i}\,b^{F}_i=\sum_{y\in\{0,1\}^m}A(y,w)\underbrace{\sum_{i=0}^{f-1}P_i(y)\,b^{F}_i}_{=\,P_\pkd(y)}=\sum_{y\in\{0,1\}^m}A(y,w)\,P_\pkd(y)\ \in\ \F_{2^f}.
\]
These are $e$ claims about the packed polynomial $P_\pkd$. Let $\ell=\lceil\log_2 e\rceil$, identify each $w<e$ with its $\ell$-bit encoding $\bar w\in\{0,1\}^{\ell}$, and pad $t_w=A(y,w)=0$ for $e\le w<2^\ell$. The verifier samples a multilinear batching point $s\leftarrow\F_{2^f}^{\ell}$ and uses weights $\lambda_w=\eq(s,\bar w)$. One sumcheck proves
\[
\sum_{y\in\{0,1\}^m}B_s(y)\,P_\pkd(y)=T(s),\qquad B_s(y):=\sum_{w<2^\ell}\eq(s,\bar w)A(y,w),\qquad T(s):=\sum_{w<2^\ell}\eq(s,\bar w)t_w.
\]
Its output is the packed claim $P_\pkd(r')=\alpha'$ at a point $r'\in\F_{2^f}^m$; the verifier computes $B_s(r')$ itself. This is the complete reduction.

\paragraph{Soundness.} Suppose the input is wrong: $\alpha_i\neq P_i(r)$ for some $i$, against the committed $P_\pkd$. Then the reduction outputs a correct packed claim $P_\pkd(r')=\alpha'$ with probability at most
\[
\underbrace{\frac{\ell}{2^f}}_{\text{batching}}\;+\;\underbrace{\frac{2m}{2^f}}_{\text{sumcheck}}\;=\;\frac{2m+\ell}{2^f}.
\]
\begin{enumerate}
\item The map $(\alpha_i)_{i=0}^{f-1}\mapsto(t_w)_{w=0}^{e-1}$ is a \emph{bijection} $\F_{2^e}^f\to\F_{2^f}^e$. Therefore a wrong input forces $t_w\neq\sum_yA(y,w)P_\pkd(y)$ for at least one $w$.
\item The multilinear extension of these coordinate errors is nonzero because it is nonzero at that Boolean point $\bar w$. By Schwartz--Zippel, its evaluation at $s\leftarrow\F_{2^f}^{\ell}$ vanishes with probability at most $\ell/2^f$.
\item Sumcheck started from this false value (summand of degree $2$, over $m$ variables) ends at a correct claim $P_\pkd(r')=\alpha'$ with probability $\le 2m/2^f$.
\end{enumerate}

\section*{Evaluating $B$}
\label{section:eval_B}

It remains to compute $B_s(r')$. The value $A(y,w)\in\F_2$ is the $w$-th coordinate of $\eq(r,y)$, i.e.\ $\eq(r,y)=\sum_{w=0}^{e-1}A(y,w)b^E_w$. Following~\cite{Flock}, define the $\F_2$-linear map $\Phi_s:\F_{2^e}\to\F_{2^f}$ by $\Phi_s(b^E_w)=\eq(s,\bar w)$. For every $y\in\{0,1\}^m$:
\[
\Phi_s\big(\eq(r,y)\big)=\Phi_s\Big(\sum_{w=0}^{e-1}A(y,w)b^E_w\Big)=\sum_{w=0}^{e-1}A(y,w)\eq(s,\bar w)=B_s(y).
\]
Thus, on the hypercube, $B_s$ is ``compute $\eq(r,y)\in\F_{2^e}$, then apply $\Phi_s$.''

\paragraph{Closed form.} Two elementary facts do all the work.

\emph{Fact 1 ($\eq$ in characteristic $2$).} In each factor of $\eq$ the two $x_jy_j$ terms cancel:
\[
\eq(x,y)=\prod_{j=1}^m\bigl(x_jy_j+(1+x_j)(1+y_j)\bigr)=\prod_{j=1}^m(1+x_j+y_j).
\]

\emph{Fact 2 (linear maps are sums of squarings).} View $\Phi_s$ as mapping into $L=\F_{2^{\mathrm{lcm}(e,f)}}$, the smallest field containing both $\F_{2^e}$ and $\F_{2^f}$ (for $e=f$: $L=\F_{2^e}$, nothing to embed). Squaring is additive in characteristic $2$, so any expression
\[
a\;\mapsto\;c_0\,a+c_1\,a^2+c_2\,a^4+\dots+c_{e-1}\,a^{2^{e-1}},\qquad c_k\in L,
\]
is an $\F_2$-linear map $\F_{2^e}\to L$, and every such map can be written this way. We find the $c_k$ by writing out what $\Phi_s$ must do on the basis:
\[
\sum_{k=0}^{e-1}\bigl(b^E_w\bigr)^{2^k}c_k=\eq(s,\bar w),\qquad w=0,\dots,e-1.
\]
This square system is invertible: a combination that vanishes on the basis vanishes on all $2^e$ field elements, whereas its degree is at most $2^{e-1}$, so all its coefficients must be zero. Its matrix depends only on the basis and can therefore be inverted once; each opening then computes $(c_k)_{k<e}$ by an $O(e^2)$ matrix-vector product.

Now compute. Squaring respects sums and products and fixes bits ($y_j^2=y_j$), so on the hypercube, by Fact~1,
\[
\eq(r,y)^{2^k}=\prod_{j=1}^m(1+r_j+y_j)^{2^k}=\prod_{j=1}^m\bigl(1+r_j^{2^k}+y_j\bigr)=\eq\bigl(r^{2^k},y\bigr),
\qquad r^{2^k}:=\bigl(r_1^{2^k},\dots,r_m^{2^k}\bigr).
\]
Since $B_s$ is multilinear, evaluating it at $r'$ and expanding with Fact~2:
\[
B_s(r')=\sum_{y\in\{0,1\}^m}B_s(y)\,\eq(r',y)
=\sum_{y\in\{0,1\}^m}\Phi_s\bigl(\eq(r,y)\bigr)\,\eq(r',y)
=\sum_{k<e}c_k\sum_{y\in\{0,1\}^m}\eq\bigl(r^{2^k},y\bigr)\,\eq(r',y).
\]
The inner sum evaluates, at $r'$, the multilinear extension of $y\mapsto\eq(r^{2^k},y)$, which is $\eq(r^{2^k},\cdot\,)$ itself. So it collapses to $\eq(r^{2^k},r')$, and by Fact~1 again:
\[
\boxed{\ \
B_s(r') \;=\; \sum_{k<e}\, c_k \prod_{j=1}^m
\bigl( 1 + r_j^{2^k} + r'_j \bigr).
\ \ }
\]

\paragraph{Cost.} Level $k$ needs one squaring per coordinate (to get $r_j^{2^k}$ from $r_j^{2^{k-1}}$) and one $m$-factor product: $O(em)$ multiplications in $L$, on top of the generic $O(e^2)$ construction of the $c_k$.

\section*{Worked example: the GHASH power basis}

For a general basis, constructing the $c_k$ after precomputing the fixed matrix inverse costs one $O(e^2)$ matrix-vector product per opening. In the GHASH field, over its power basis, the coefficients instead have a closed form using $15$ fixed seed constants and $O(\log e)$ multiplications per coefficient.

Let $e=f=128$, so $L=\F_{2^{128}}$ and no embedding is involved. Define
\[
p(X)=X^{128}+X^7+X^2+X+1,
\qquad
\F_{2^{128}}=\F_2[X]/\bigl(p(X)\bigr),
\]
and let $\alpha$ denote the residue of $X$ modulo $p$. The basis is $b^E_i=\alpha^i$, $i<128$. Then $\ell=7$, the batching point is $s=(s_0,\dots,s_6)$, and the weights are $\lambda_i=\eq(s,\bar\imath)$.

\paragraph{Step 1: extracting binary coordinates.} Define $\operatorname{tr}(z)=\sum_{k<128}z^{2^k}$, an $\F_2$-valued linear function. We will construct constants $\delta_i$ satisfying
\[
\operatorname{tr}\bigl(\delta_i\alpha^j\bigr)=[\,i=j\,].
\]
Consequently, if $x=\sum_{i<128}x_i\alpha^i$ with bits $x_i\in\F_2$, then $\operatorname{tr}(\delta_ix)=x_i$: multiplying by $\delta_i$ and applying $\operatorname{tr}$ extracts the $i$-th binary coordinate. Since $\Phi_s(\alpha^i)=\lambda_i$ by definition and $\Phi_s$ is $\F_2$-linear,
\[
\Phi_s(x)=\sum_{i<128}\lambda_ix_i=\sum_{i<128}\lambda_i\operatorname{tr}(\delta_ix)
=\sum_{k<128}\Bigl(\underbrace{\textstyle\sum_{i<128}\lambda_i\delta_i^{2^k}}_{=\,c_k}\Bigr)x^{2^k}.
\]
Thus the coefficients are $c_k=\sum_{i<128}\lambda_i\delta_i^{2^k}$.

\paragraph{Step 2: constructing the coordinate extractors.} Let $d=p'(\alpha)=\alpha^6+1$; only the odd-degree terms $X^7$ and $X$ of $p$ survive differentiation in characteristic $2$. The roots of $p$ are $\alpha,\alpha^2,\ldots,\alpha^{2^{127}}$. Partial fractions, followed by expanding each $(X+\beta)^{-1}$ in powers of $X^{-1}$, gives
\[
\frac{1}{p(X)}
=\sum_{j=0}^{127}\frac{1}{p'(\alpha^{2^j})(X+\alpha^{2^j})}
=\sum_{k\ge0}\operatorname{tr}\!\Bigl(\frac{\alpha^k}{p'(\alpha)}\Bigr)X^{-(k+1)}.
\]
Here the final sum is exactly a trace because $p'(\alpha^{2^j})=p'(\alpha)^{2^j}$. We can therefore read $\operatorname{tr}(\alpha^k/d)$ from the coefficient of $X^{-(k+1)}$ in $1/p$. Directly from the sparse polynomial,
\[
\frac1{p(X)}=X^{-128}(1+u)^{-1}=X^{-128}\bigl(1+u+u^2+\cdots\bigr),
\qquad
u=X^{-121}+X^{-126}+X^{-127}+X^{-128}.
\]
The contribution $X^{-128}u^2$ starts at $X^{-370}$, corresponding to $k=369$, so only $1$ and $u$ contribute for $k\le255$:
\[
\operatorname{tr}\bigl(\alpha^k/d\bigr)=1
\iff
k\in\{127\}\cup\bigl(127+\{121,126,127,128\}\bigr)=\{127,248,253,254,255\}.
\]
So $\alpha^{127-i}/d$ is \emph{almost} $\delta_i$: it satisfies the coordinate-extraction relation at every $j\le i$, where $127-i+j\le127$, and fails only at the $j$ with $127-i+j\in\{248,253,254\}$, i.e.\ $j-i\in\{121,126,127\}$ ($j-i=128$ being out of range). Each failing $j$ is $\ge121$, so $m:=127-j$ satisfies $m\le6$; for such $m$ and any $j'<128$ we have $m+j'<248$, hence $\operatorname{tr}(\alpha^{m+j'}/d)=[\,m+j'=127\,]$. Adding $d^{-1}\alpha^{m}$ therefore flips the relation at $j'=j$ and leaves every other $j'$ untouched. Summing one such monomial per failing $j$ repairs all of them at once:
\[
\boxed{\ \delta_i=d^{-1}\bigl(\alpha^{127-i}+\varepsilon_i\bigr),\ }
\qquad
\begin{aligned}
&\varepsilon_0=\alpha^6+\alpha+1, &&\varepsilon_1=\alpha^5+1,\\
&\varepsilon_i=\alpha^{6-i}\quad(2\le i\le6), &&\varepsilon_i=0\quad(i\ge7).
\end{aligned}
\]
The corrections die out at $i=6$ because $j=i+121<128$ forces $i\le6$ (and $j=i+126$, $j=i+127$ force $i\le1$ and $i=0$, which is why $\varepsilon_0,\varepsilon_1$ carry extra monomials).

\paragraph{Step 3: factoring the $128$-term sum.} Write the index in bits, $i=\sum_{t<7}2^ti_t$. Then $127-i=\sum_{t<7}2^t(1-i_t)$, and $\lambda_i=\prod_{t<7}(1+s_t+i_t)$ by Fact~1. Both are products over $t$, so the sum over $i$ splits coordinatewise:
\[
\sum_{i<128}\lambda_i\,\alpha^{(127-i)2^k}
=\prod_{t=0}^{6}\Bigl(s_t+(1+s_t)\,\alpha^{2^{k+t}}\Bigr)
=\alpha^{127\cdot2^k}\prod_{t=0}^{6}\Bigl(1+s_t+s_t\,\alpha^{-2^{k+t}}\Bigr),
\]
the last step by $\alpha^{127\cdot2^k}=\prod_{t<7}\alpha^{2^{k+t}}$. Feeding Step 2 into Step 1 and applying this to the $\alpha^{127-i}$ part (the $\varepsilon_i$ part has only seven nonzero addends) gives
\[
\boxed{\ \
c_k=d^{-2^k}\Bigl[\
\alpha^{127\cdot2^k}\prod_{t=0}^{6}\bigl(1+s_t+s_t\,\alpha^{-2^{k+t}}\bigr)
\ +\ \sum_{i=0}^{6}\lambda_i\,\varepsilon_i^{2^k}
\ \Bigr].
\ \ }
\]

\paragraph{Cost.} The $k=0$ level needs $15$ constants: $d^{-1}\alpha^{127}$, the seven $1+\alpha^{-2^t}$, and the seven $d^{-1}\varepsilon_i$. Squaring all $15$ advances level $k$ to level $k+1$, so a verifier running a fixed program can bake in every level; and the $7\cdot128$ factor constants take only $128$ distinct values, since $1+\alpha^{-2^{k+t}}$ depends on $k+t\bmod128$ alone. Each $c_k$ then costs $14$ multiplications for the product and $7$ for the correction, so the whole table costs $O(e\log e)$ instead of $O(e^2)$. The seven weights $\lambda_i$, $i\le6$, share the four factors $1+s_t$, $t\ge3$, so they cost $O(\log e)$ as well.

\section*{Recursion specialization: univariate batching over a tower}

The previous section is only a worked example. Its closed form leans on two properties: the basis is a \emph{power} basis $\alpha^i$ of a field defined by a sparse modulus, and the weights $\lambda_i$ are $\eq$ weights, hence a product over the bits of $i$. Neither survives verbatim when the opening field is a tower, but the technique does. The recursion specialization uses $f=64$ and $e=192$ with
\[
K=\F_{2^{64}}=\F_2[X]/\bigl(q(X)\bigr),\qquad q(X)=X^{64}+X^4+X^3+X+1,
\qquad
\F_{2^{192}}=K[Y]/\bigl(Y^3+Y+1\bigr),
\]
and the product basis $b^E_{64j+i}=x^iy^j$, where $x,y$ are the residues of $X,Y$. It also batches coordinates with the univariate weights $1,\rho,\ldots,\rho^{191}$ for $\rho\in\F_{2^{192}}$. This turns the reduction directly into one weighted PCS claim; no separate ring-switch sumcheck is required. Two things change in the coefficient computation.

\paragraph{Change 1: the basis factors instead of being a power basis.} Let $\beta_i\in K$ extract coordinate $i$ of the basis $1,x,\ldots,x^{63}$ through the trace to $\F_2$, and let $\gamma_j\in\F_{2^{192}}$ similarly extract coordinate $j$ of $1,y,y^2$ through the trace to $K$. Their products $\beta_i\gamma_j$ extract the coordinates of the product basis $x^iy^j$. With univariate weights $\rho^w$ the coefficient sum therefore \emph{splits}:
\[
c_k=\sum_{w<192}\rho^w\delta_w^{2^k}
=\underbrace{\Bigl(\sum_{i<64}\rho^i\beta_i^{2^k}\Bigr)}_{A_k}
\cdot
\underbrace{\Bigl(\sum_{j<3}\rho^{64j}\gamma_j^{2^k}\Bigr)}_{B_k}.
\]
Here $\gamma_0=1$: $\operatorname{Tr}_{\F_{2^e}/K}(1)=3=1$, while $\operatorname{Tr}(y)$ and $\operatorname{Tr}(y^2)$ are symmetric functions of the roots of $Y^3+Y+1$ and vanish. So $\beta_i\in K$, and $A_k$ has Frobenius period $64$: only $64$ of the $192$ levels are distinct. $B_k$ is already two multiplications.

\paragraph{Change 2: a geometric series replaces the $\eq$ product.} Step 2 of the previous section is generic in the modulus. Here $d=q'(x)=x^2+1$, and the Laurent expansion of $1/q$ at infinity is $X^{-64}+\bigl(X^{-124}+X^{-125}+X^{-127}+X^{-128}\bigr)+O(X^{-184})$, so $\operatorname{tr}(x^m/d)=[\,m=63\,]$ for all $m<120$ and $x^{63-i}/d$ fails only at $j-i\in\{60,61,63\}$. Repairing exactly as before,
\[
\beta_i=d^{-1}\bigl(x^{63-i}+\varepsilon_i\bigr),
\qquad
\varepsilon=\bigl[\,x^3+x^2+1,\ x^2+x,\ x+1,\ 1,\ 0,\dots,0\,\bigr],
\]
a correction on $i<4$ only (versus seven for GHASH: the modulus is sparser relative to the degree).

Step 3 is where the weight family matters. The $\lambda_i$ no longer factor over the bits of $i$, so there is no $6$-fold product. But univariate weights give something equally good: with $u=\rho\,x^{-2^k}$,
\[
\sum_{i<64}\rho^i x^{(63-i)2^k}=x^{63\cdot2^k}\sum_{i<64}u^i
=x^{63\cdot2^k}\,(1+u)^{63},
\]
because $\sum_{i<n}u^i=(u^n+1)/(u+1)$ and $n=64$ is a power of two, so $u^{64}+1=(u+1)^{64}$. The identity holds for \emph{every} $u$, including $u=1$, where both sides are $0$. Hence
\[
\boxed{\ \
A_k=Q_k\,\bigl(1+\rho P_k\bigr)^{63}+\sum_{i<4}\rho^i R_{k,i},
\qquad
P_k=x^{-2^k},\quad Q_k=\bigl(d^{-1}x^{63}\bigr)^{2^k},\quad R_{k,i}=\bigl(d^{-1}\varepsilon_i\bigr)^{2^k}.
\ \ }
\]

\paragraph{Cost.} Level $k$ needs $6$ constants, and squaring the row advances $k$ to $k+1$, so a fixed-program verifier bakes in all $64$ base levels and all $192$ tower levels and keeps no runtime orbit table. Evaluating $(1+u)^{63}$ by the $2^t-1$ addition chain costs $5$ squarings and $3$ products, so $A_k$ costs $13$ multiplications and $c_k$ costs $16$: the same $O(\log f)$ per coefficient the $\eq$ product achieves, reached through a different identity. The whole table drops from $64^2+3\cdot192$ multiplications to $64\cdot13+192\cdot3$.

\paragraph{What generalizes.} Only Step 3 is weight-specific, and it needs the index sum to collapse. Two sufficient conditions, both met here: the basis is a power basis of the field the weights are indexed over (so $b_{n-1-i}$ is a geometric progression in $i$), and the weight family is closed under the corresponding collapse -- a product over bits for $\eq$, a geometric series for univariate powers. Steps 1 and 2 need only a sparse modulus, and the tower factorization of Change 1 applies to any product basis.

\section*{Remarks}

\begin{enumerate}
\item \emph{Small $\F_{2^f}$.} If $2^f$ is too small for the error $(2m+\ell)/2^f$, sample the batching point $s$ (and run the sumcheck) in an extension $L\supseteq\F_{2^f}$: the error becomes $(2m+\ell)/|L|$, and $P_\pkd$ is just opened at $r'\in L^m$.
\item \emph{Other characteristics.} The coordinate-batching reduction extends to other base fields, but the binary closed forms above use characteristic two essentially: squaring is the relevant Frobenius map and $\eq(x,y)$ simplifies to $\prod_j(1+x_j+y_j)$. Other characteristics require their corresponding linearized maps and equality polynomial.
\end{enumerate}
